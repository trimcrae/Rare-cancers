def wilson_width(p, n, z=1.959963984540054):
    d = 1 + z*z/n
    half = (z/d)*math.sqrt(p*(1-p)/n + z*z/(4*n*n))
    return 2*half

def n_for_width(p, target=0.20):

    while n < 100000:
        if wilson_width(p, n) <= target:

        n += 1
    return None
